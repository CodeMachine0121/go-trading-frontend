// 前言：輸入與參數（可編輯）
package strategy
func Signal(candles []KCandle, p Params) Decision {
lookback := p.Int("lookback", 20)
multiplier := p.Decimal("atrMultiplier", 1.5)
window := candles[len(candles)-lookback:]
high := highestHigh(window)
last := candles[len(candles)-1]
if last.Close.GreaterThan(high) {
stop := last.Close.Sub(atr(candles).Mul(multiplier))
return Enter(Long).StopLoss(stop)
}
return Hold()
}
